cmake_minimum_required(VERSION 3.24)
project(PulseText VERSION 1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "PulseText plugins require Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  add_compile_options(/utf-8)
endif()
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d)
FetchContent_MakeAvailable(json)
set(FFGL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/vendor/ffgl" CACHE PATH "Official FFGL 2.2 checkout")
file(GLOB SDK CONFIGURE_DEPENDS "${FFGL_ROOT}/source/lib/ffgl/*.cpp" "${FFGL_ROOT}/source/lib/ffglex/*.cpp")
foreach(kind Source Effect)
  add_library(PulseText${kind} SHARED ${SDK} native/Plugin.cpp native/Receiver.cpp native/Text.cpp
    "${FFGL_ROOT}/source/lib/glsdk_0_5_2/glload/source/gl_load.c"
    "${FFGL_ROOT}/source/lib/glsdk_0_5_2/glload/source/wgl_load.c")
  target_include_directories(PulseText${kind} PRIVATE "${FFGL_ROOT}/source/lib" "${FFGL_ROOT}/source/lib/glsdk_0_5_2/glload/include")
  target_compile_definitions(PulseText${kind} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
  set_target_properties(PulseText${kind} PROPERTIES PREFIX "")
  if(MINGW)
    target_link_options(PulseText${kind} PRIVATE -static)
  endif()
  if(kind STREQUAL "Effect")
    target_compile_definitions(PulseText${kind} PRIVATE PULSE_EFFECT=1)
  endif()
  target_link_libraries(PulseText${kind} PRIVATE nlohmann_json::nlohmann_json opengl32 ws2_32 dwrite d2d1 windowscodecs ole32)
  install(TARGETS PulseText${kind} RUNTIME DESTINATION .)
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(native_receiver tests/native_receiver.cpp native/Receiver.cpp)
  target_include_directories(native_receiver PRIVATE native)
  target_link_libraries(native_receiver PRIVATE nlohmann_json::nlohmann_json ws2_32)
  target_compile_definitions(native_receiver PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  add_executable(native_text tests/native_text.cpp native/Text.cpp)
  target_include_directories(native_text PRIVATE native)
  target_compile_definitions(native_text PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(native_text PRIVATE nlohmann_json::nlohmann_json dwrite d2d1 windowscodecs ole32)
  if(MINGW)
    target_link_options(native_receiver PRIVATE -static)
    target_link_options(native_text PRIVATE -static)
  endif()
  add_test(NAME watchdog COMMAND native_receiver --timeout)
  add_test(NAME text_raster COMMAND native_text)
  add_executable(native_gl tests/native_gl.cpp "${FFGL_ROOT}/source/lib/glsdk_0_5_2/glload/source/gl_load.c")
  target_include_directories(native_gl PRIVATE "${FFGL_ROOT}/source/lib")
  target_link_libraries(native_gl PRIVATE nlohmann_json::nlohmann_json opengl32 gdi32 user32 ws2_32)
  if(MINGW)
    target_link_options(native_gl PRIVATE -static)
  endif()
  add_test(NAME source_gl COMMAND native_gl $<TARGET_FILE:PulseTextSource>)
  add_test(NAME effect_gl COMMAND native_gl $<TARGET_FILE:PulseTextEffect> effect)
  set_tests_properties(source_gl effect_gl PROPERTIES RUN_SERIAL TRUE)
  set_tests_properties(source_gl PROPERTIES ENVIRONMENT "PULSETEXT_SOURCE_PORT=12400")
  set_tests_properties(effect_gl PROPERTIES ENVIRONMENT "PULSETEXT_EFFECT_PORT=12401")
endif()
